// Behavioural: false means a wholesale swap to spackle's Promise. A native that
// only lacks finally/allSettled/any must pass; those islands patch it per method.
// Two checks an old native fails:
//   1. PromiseRejectionEvent (Chrome 49+ / Safari 11+). Without it unhandled
//      rejections are silent; install() below wires the event.
//   2. @@species: `p.then()` must construct through the species constructor.
import { define } from '../define.js';

export const isSupported = (): boolean => {
  try {
    if (
      typeof Promise !== 'function' ||
      typeof Promise.prototype.then !== 'function' ||
      typeof Promise.prototype.catch !== 'function' ||
      typeof Promise.resolve !== 'function' ||
      typeof Promise.reject !== 'function'
    ) {
      return false;
    }

    // Already ours. Re-installing would swap a live constructor
    // out from under every promise already created against it.
    if ((Promise as any).__polyfilled === true) {
      return true;
    }

    const w: any = window;
    if (typeof w.PromiseRejectionEvent !== 'function') {
      return false;
    }

    // No @@species at all (Chrome 41 predates it) means the engine cannot
    // honour it, so the check is unrunnable rather than passing by default.
    const species = typeof Symbol !== 'undefined' ? (Symbol as any).species : undefined;
    if (!species) {
      return false;
    }

    const Fake: any = function (executor: any): void {
      executor(function (): void { /* resolve */ }, function (): void { /* reject */ });
    };
    const probe: any = new Promise(function (resolve: any): void {
      resolve(1);
    });
    probe.constructor = {};
    probe.constructor[species] = Fake;
    return probe.then(function (): void { /* species probe */ }) instanceof Fake;
  } catch {
    return false;
  }
};

/**
 * Spec: https://tc39.es/ecma262/#sec-promise-objects
 *
 * spackle's own Promise, written to the spec algorithms: NewPromiseCapability
 * and SpeciesConstructor everywhere, thenables resolved in a job, resolving
 * functions that are not constructors (where the engine parses arrows), and
 * HostPromiseRejectionTracker driving unhandledrejection / rejectionhandled with
 * the promise attached.
 *
 * Jobs: one microtask per job through the engine's own queueMicrotask (or a
 * native promise's then), so they interleave with native microtasks exactly as
 * native promise jobs would. With neither (Safari 7.1), a batched
 * MutationObserver queue -- see scheduleDrain below.
 *
 * Known limitations:
 * - [incomplete] Where the engine has no arrow functions, resolving and element
 *   functions are ordinary functions: they have a prototype and can be `new`ed.
 */

const hasOwn = Object.prototype.hasOwnProperty;
const isObject = (value: any): boolean =>
  value !== null && (typeof value === 'object' || typeof value === 'function');
const isConstructor = (C: any): boolean =>
  typeof C === 'function' && (hasOwn.call(C, 'prototype') || /^bound /.test(C.name || ''));

// Built-in closures are not constructors and have no prototype: an arrow, where
// the engine parses one. new Function only here, at module evaluation.
let arrowWrap: any = null;
try {
  arrowWrap = new Function('f', 'return (x) => f(x)');
} catch (e) {
  /* no arrow syntax (or CSP): plain functions */
}
const builtinClosure = (fn: (x: any) => any): any =>
  arrowWrap ? arrowWrap(fn) : function (x: any) { return fn(x); };

/* ---------------------------------------------------------------- jobs */

/*
 * The job queue, flat: [kind, a, b, c] per job, so queueing allocates nothing
 * beyond the array growth. Kinds: 0 PromiseReactionJob(reaction, argument),
 * 1 PromiseResolveThenableJob(promise, thenable, then).
 */
const jobs: any[] = [];
let jobHead = 0;

const runJob = (kind: number, a: any, b: any, c: any): void => {
  if (kind === 0) promiseReactionJob(a, b);
  else promiseResolveThenableJob(a, b, c);
};

// Run exactly one queued job: what each host microtask does.
const runOne = function (): void {
  const kind = jobs[jobHead];
  const a = jobs[jobHead + 1];
  const b = jobs[jobHead + 2];
  const c = jobs[jobHead + 3];
  jobs[jobHead] = jobs[jobHead + 1] = jobs[jobHead + 2] = jobs[jobHead + 3] = undefined;
  jobHead += 4;
  if (jobHead === jobs.length) {
    jobs.length = 0;
    jobHead = 0;
  } else if (jobHead > 4096) {
    // slide the unrun tail down so a long-lived queue doesn't grow forever
    const rest = jobs.length - jobHead;
    for (let i = 0; i < rest; i++) jobs[i] = jobs[jobHead + i];
    jobs.length = rest;
    jobHead = 0;
  }
  runJob(kind, a, b, c);
};

const w0: any = typeof window !== 'undefined' ? window : {};
// Captured before install replaces anything.
const hostQueueMicrotask: any = typeof w0.queueMicrotask === 'function' ? w0.queueMicrotask : null;
const NativePromise: any = typeof w0.Promise === 'function' && w0.Promise.__polyfilled !== true ? w0.Promise : null;
const nativeResolved: any = NativePromise ? NativePromise.resolve() : null;
const nativeThen: any = NativePromise ? NativePromise.prototype.then : null;

let batched: ((fn: () => void) => void) | null = null;
const batchedRun = function (): void {
  runOne();
};

const enqueueJob = (kind: number, a: any, b: any, c: any): void => {
  jobs.push(kind, a, b, c);
  if (hostQueueMicrotask) {
    hostQueueMicrotask.call(w0, runOne);
  } else if (nativeThen) {
    nativeThen.call(nativeResolved, runOne);
  } else {
    if (batched === null) batched = scheduleDrain();
    batched(batchedRun);
  }
};

/*
 * The fallback scheduler, for engines with no microtask of their own. A timer
 * per job would clamp to 4ms past depth 5, so long .then chains took seconds.
 * The shape is core-js's (MIT):
 * 1. Batch: one wake-up per empty -> non-empty transition; the drain runs until
 *    the queue is empty, including callbacks queued during the drain.
 * 2. MutationObserver (a microtask: unclamped, before paint), MessageChannel as
 *    the fallback. The observer exclusions are core-js's: some old iOS and webOS
 *    WebKit builds don't fire it reliably.
 */
const scheduleDrain = (): ((fn: () => void) => void) => {
  const queue: any[] = [];
  // Index, not shift() (O(n) on old engines); slots are nulled as consumed.
  let head = 0;
  let scheduled = false;
  let wake: () => void;

  const drain = (): void => {
    scheduled = false;
    while (head < queue.length) {
      const fn = queue[head];
      queue[head] = undefined;
      head++;
      try {
        fn();
      } catch (error) {
        // Rethrow from a timer: the error still reaches window.onerror without
        // abandoning the reactions queued behind it.
        w.setTimeout(function () {
          throw error;
        }, 0);
      }
      // Past 1024, slide the unrun tail down to 0 so a long-lived drain doesn't grow
      // the array forever (asap's fix). Amortised O(1).
      if (head > 1024) {
        const rest = queue.length - head;
        for (let i = 0; i < rest; i++) queue[i] = queue[head + i];
        queue.length = rest;
        head = 0;
      }
    }
    queue.length = 0;
    head = 0;
  };

  const w: any = window;
  const MO = w.MutationObserver || w.WebKitMutationObserver;
  const ua = (w.navigator && w.navigator.userAgent) || '';
  const brokenObserver = /(?:ipad|iphone|ipod).*applewebkit/i.test(ua) || /web0s(?!.*chrome)/i.test(ua);

  /*
   * Backup timer alongside the fast path, only where the fast path can stall: a
   * script-implemented MutationObserver built on the global Promise (jsdom)
   * deadlocks once that Promise is this polyfill. Whichever fires first drains.
   */
  let fast: () => void;
  // A native MutationObserver can't stall, so it skips the timer.
  // MessageChannel keeps it, as in install-set-immediate.
  let needsBackup = true;

  if (MO && w.document && !brokenObserver) {
    // Detached text node: flipping its data is the cheapest observable mutation.
    let toggle = true;
    const node = w.document.createTextNode('');
    new MO(drain).observe(node, { characterData: true });
    try {
      needsBackup = !/\[native code\]/.test(Function.prototype.toString.call(MO));
    } catch (e) {
      /* unreadable source: keep the timer */
    }
    fast = function (): void {
      toggle = !toggle;
      node.data = toggle ? '1' : '0';
    };
  } else if (typeof w.MessageChannel === 'function') {
    const channel = new w.MessageChannel();
    channel.port1.onmessage = drain;
    fast = function (): void {
      channel.port2.postMessage(0);
    };
  } else {
    fast = function (): void {};
  }

  wake = function (): void {
    fast();
    if (needsBackup) w.setTimeout(drain, 0);
  };

  return function (fn: () => void): void {
    queue.push(fn);
    if (scheduled) return;
    scheduled = true;
    wake();
  };
};

/* ------------------------------------------------- rejection tracking */

/*
 * Report an unhandled rejection the way the platform does: isSupported() rejects
 * natives without PromiseRejectionEvent, so this must provide one.
 * createEvent/initEvent because these engines lack the event constructors.
 * onunhandledrejection is called explicitly only where the engine doesn't wire
 * it as an event-handler attribute (else every rejection arrives twice); checked
 * at install time, before app code can assign it.
 */
let handlerAttributeIsWired = false;

const dispatchRejectionEvent = function (type: string, promise: any, reason: any): boolean {
  const w: any = window;
  let prevented = false;
  try {
    const doc = w.document;
    if (doc && typeof doc.createEvent === 'function' && typeof w.dispatchEvent === 'function') {
      const event: any = doc.createEvent('Event');
      event.initEvent(type, false, type === 'unhandledrejection');
      event.reason = reason;
      event.promise = promise;
      // dispatchEvent returns false when a listener cancelled the event.
      prevented = w.dispatchEvent(event) === false;
      const handler = w['on' + type];
      if (!handlerAttributeIsWired && typeof handler === 'function') {
        handler.call(w, event);
        prevented = event.defaultPrevented === true;
      }
    } else if (typeof w['on' + type] === 'function') {
      // No DOM to build an event with -- hand over a shaped object instead.
      w['on' + type]({ reason: reason, promise: promise });
    }
  } catch (e) {
    // Reporting a failure must never throw over the failure it is reporting.
  }
  return prevented;
};

export const reportUnhandledRejection = function (reason: any, promise?: any): void {
  const prevented = dispatchRejectionEvent('unhandledrejection', promise === undefined ? null : promise, reason);
  if (!prevented && typeof console !== 'undefined' && console.warn) {
    console.warn('Unhandled promise rejection:', reason);
  }
};

/*
 * HostPromiseRejectionTracker. Rejections without a handler are collected and
 * checked after the current task (a timer), as browsers do at the end of a
 * microtask checkpoint; one handled later than that gets rejectionhandled.
 */
const pendingRejections: any[] = [];
let rejectionCheckScheduled = false;
const checkRejections = function (): void {
  rejectionCheckScheduled = false;
  const list = pendingRejections.splice(0, pendingRejections.length);
  for (let i = 0; i < list.length; i++) {
    const promise = list[i];
    if (promise._isHandled) continue;
    promise._reported = true;
    reportUnhandledRejection(promise._result, promise);
  }
};
const trackRejection = (promise: any, operation: 'reject' | 'handle'): void => {
  if (operation === 'reject') {
    pendingRejections.push(promise);
    if (!rejectionCheckScheduled) {
      rejectionCheckScheduled = true;
      setTimeout(checkRejections, 0);
    }
  } else if (promise._reported) {
    promise._reported = false;
    setTimeout(function () {
      dispatchRejectionEvent('rejectionhandled', promise, promise._result);
    }, 0);
  }
};

/* --------------------------------------------------------- the Promise */

const PENDING = 0;
const FULFILLED = 1;
const REJECTED = 2;
const SLOT: PropertyDescriptor = { value: undefined, writable: true, enumerable: false, configurable: false };
const defineSlot = (target: any, key: string, value: any): void => {
  SLOT.value = value;
  Object.defineProperty(target, key, SLOT);
  SLOT.value = undefined;
};
// IsPromise: an object carrying this module's [[PromiseState]] slot.
const isPromise = (x: any): boolean => isObject(x) && hasOwn.call(x, '_state');

const triggerReactions = (reactions: any[], argument: any): void => {
  for (let i = 0; i < reactions.length; i++) enqueueJob(0, reactions[i], argument, undefined);
};

const fulfillPromise = (promise: any, value: any): void => {
  const reactions = promise._fulfillReactions;
  promise._result = value;
  promise._fulfillReactions = undefined;
  promise._rejectReactions = undefined;
  promise._state = FULFILLED;
  triggerReactions(reactions, value);
};

const rejectPromise = (promise: any, reason: any): void => {
  const reactions = promise._rejectReactions;
  promise._result = reason;
  promise._fulfillReactions = undefined;
  promise._rejectReactions = undefined;
  promise._state = REJECTED;
  if (!promise._isHandled) trackRejection(promise, 'reject');
  triggerReactions(reactions, reason);
};

// The steps of a promise resolve function, after the alreadyResolved check.
const resolveSteps = (promise: any, resolution: any): void => {
  if (resolution === promise) {
    rejectPromise(promise, new TypeError('Chaining cycle detected for promise'));
    return;
  }
  if (!isObject(resolution)) {
    fulfillPromise(promise, resolution);
    return;
  }
  let then: any;
  try {
    then = resolution.then;
  } catch (error) {
    rejectPromise(promise, error);
    return;
  }
  if (typeof then !== 'function') {
    fulfillPromise(promise, resolution);
    return;
  }
  // NewPromiseResolveThenableJob: the thenable's then runs in a later job
  enqueueJob(1, promise, resolution, then);
};

// CreateResolvingFunctions: a resolve/reject pair sharing alreadyResolved.
const createResolvingFunctions = (promise: any): any => {
  const state = { resolved: false };
  return {
    resolve: builtinClosure(function (resolution: any) {
      if (state.resolved) return undefined;
      state.resolved = true;
      resolveSteps(promise, resolution);
      return undefined;
    }),
    reject: builtinClosure(function (reason: any) {
      if (state.resolved) return undefined;
      state.resolved = true;
      rejectPromise(promise, reason);
      return undefined;
    }),
  };
};

function promiseReactionJob(reaction: any, argument: any): void {
  const capability = reaction.capability;
  const handler = reaction.handler;
  let result: any;
  let abrupt = false;
  if (handler === undefined) {
    result = argument;
    abrupt = reaction.type === REJECTED;
  } else {
    try {
      result = handler(argument);
    } catch (error) {
      result = error;
      abrupt = true;
    }
  }
  if (capability === undefined) return;
  if (abrupt) capability.reject(result);
  else capability.resolve(result);
}

function promiseResolveThenableJob(promise: any, thenable: any, then: any): void {
  const resolving = createResolvingFunctions(promise);
  try {
    then.call(thenable, resolving.resolve, resolving.reject);
  } catch (error) {
    resolving.reject(error);
  }
}

// Slots for a new promise: [[PromiseState]] and friends, non-enumerable.
const initializePromise = (promise: any): void => {
  defineSlot(promise, '_state', PENDING);
  defineSlot(promise, '_result', undefined);
  defineSlot(promise, '_fulfillReactions', []);
  defineSlot(promise, '_rejectReactions', []);
  defineSlot(promise, '_isHandled', false);
  defineSlot(promise, '_reported', false);
};

export const PromisePolyfill: any = function Promise(this: any, executor: any) {
  if (!(this instanceof PromisePolyfill)) {
    throw new TypeError("Promise constructor cannot be invoked without 'new'");
  }
  if (typeof executor !== 'function') {
    throw new TypeError('Promise resolver ' + (isObject(executor) ? 'is not a function' : String(executor) + ' is not a function'));
  }
  initializePromise(this);
  const resolving = createResolvingFunctions(this);
  try {
    executor(resolving.resolve, resolving.reject);
  } catch (error) {
    resolving.reject(error);
  }
};

/*
 * NewPromiseCapability(C). For this module's own constructor the promise and its
 * resolving functions are made directly (same result, no executor closure).
 */
const newPromiseCapability = (C: any): any => {
  if (C === PromisePolyfill) {
    const promise = Object.create(PromisePolyfill.prototype);
    initializePromise(promise);
    const resolving = createResolvingFunctions(promise);
    return { promise: promise, resolve: resolving.resolve, reject: resolving.reject };
  }
  if (!isConstructor(C)) throw new TypeError('Promise capability: not a constructor');
  const capability: any = { promise: undefined, resolve: undefined, reject: undefined };
  const executor = function (resolve: any, reject: any): void {
    if (capability.resolve !== undefined) throw new TypeError('Promise executor has already been invoked with non-undefined arguments');
    if (capability.reject !== undefined) throw new TypeError('Promise executor has already been invoked with non-undefined arguments');
    capability.resolve = resolve;
    capability.reject = reject;
  };
  capability.promise = new C(executor);
  if (typeof capability.resolve !== 'function') throw new TypeError('Promise resolve function is not callable');
  if (typeof capability.reject !== 'function') throw new TypeError('Promise reject function is not callable');
  return capability;
};

const performPromiseThen = (promise: any, onFulfilled: any, onRejected: any, capability: any): any => {
  const fulfillReaction = { capability: capability, type: FULFILLED, handler: typeof onFulfilled === 'function' ? onFulfilled : undefined };
  const rejectReaction = { capability: capability, type: REJECTED, handler: typeof onRejected === 'function' ? onRejected : undefined };
  if (promise._state === PENDING) {
    promise._fulfillReactions.push(fulfillReaction);
    promise._rejectReactions.push(rejectReaction);
  } else if (promise._state === FULFILLED) {
    enqueueJob(0, fulfillReaction, promise._result, undefined);
  } else {
    if (!promise._isHandled) trackRejection(promise, 'handle');
    enqueueJob(0, rejectReaction, promise._result, undefined);
  }
  promise._isHandled = true;
  return capability === undefined ? undefined : capability.promise;
};

// SpeciesConstructor(O, %Promise%)
const speciesConstructor = (O: any): any => {
  const C = O.constructor;
  if (C === undefined) return PromisePolyfill;
  if (!isObject(C)) throw new TypeError('Promise constructor is not an object');
  const speciesKey = typeof Symbol !== 'undefined' ? (Symbol as any).species : undefined;
  const S = speciesKey ? C[speciesKey] : undefined;
  if (S === undefined || S === null) return PromisePolyfill;
  if (isConstructor(S)) return S;
  throw new TypeError('Promise @@species is not a constructor');
};

const promiseThen = function then(this: any, onFulfilled: any, onRejected: any): any {
  const promise = this;
  if (!isPromise(promise)) throw new TypeError('Method Promise.prototype.then called on incompatible receiver');
  const C = speciesConstructor(promise);
  const capability = newPromiseCapability(C);
  return performPromiseThen(promise, onFulfilled, onRejected, capability);
};

const promiseCatch = function (this: any, onRejected: any): any {
  // Invoke(promise, "then", « undefined, onRejected »): generic on purpose
  const promise = this;
  if (promise === null || promise === undefined) throw new TypeError('Promise.prototype.catch called on null or undefined');
  return promise.then(undefined, onRejected);
};

// PromiseResolve(C, x)
const promiseResolve = (C: any, x: any): any => {
  if (isPromise(x) && x.constructor === C) return x;
  const capability = newPromiseCapability(C);
  capability.resolve(x);
  return capability.promise;
};

const staticResolve = function resolve(this: any, x: any): any {
  if (!isObject(this)) throw new TypeError('PromiseResolve called on non-object');
  return promiseResolve(this, x);
};

const staticReject = function reject(this: any, r: any): any {
  const capability = newPromiseCapability(this);
  capability.reject(r);
  return capability.promise;
};

/* Promise.all and Promise.race: GetPromiseResolve before GetIterator, the
 * iterator protocol (array-likes on Symbol-less engines), close on our-side
 * failures, internal lists that no prototype setter can observe. */
const closeIterator = (iterator: any): void => {
  try {
    const ret = iterator['return'];
    if (ret !== undefined && ret !== null) ret.call(iterator);
  } catch (e) {
    /* the original error wins */
  }
};
const DONE = {};
const getIteratorRecord = (iterable: any): any => {
  const key = typeof Symbol !== 'undefined' ? (Symbol as any).iterator : undefined;
  if (key) {
    if (iterable === null || iterable === undefined) throw new TypeError('object is not iterable');
    const method = iterable[key];
    if (typeof method !== 'function') throw new TypeError('object is not iterable');
    const iterator = method.call(iterable);
    if (!isObject(iterator)) throw new TypeError('Result of the Symbol.iterator method is not an object');
    return { iterator: iterator, next: iterator.next, arrayLike: null, index: 0, done: false };
  }
  if (iterable === null || iterable === undefined || typeof iterable.length !== 'number') {
    throw new TypeError('object is not iterable');
  }
  return { iterator: null, next: null, arrayLike: iterable, index: 0, done: false };
};
const stepValue = (record: any): any => {
  if (record.arrayLike !== null) {
    if (record.index >= record.arrayLike.length) {
      record.done = true;
      return DONE;
    }
    return record.arrayLike[record.index++];
  }
  try {
    const result = record.next.call(record.iterator);
    if (!isObject(result)) throw new TypeError('Iterator result is not an object');
    if (result.done) {
      record.done = true;
      return DONE;
    }
    return result.value;
  } catch (error) {
    record.done = true;
    throw error;
  }
};
const LIST: PropertyDescriptor = { value: undefined, writable: true, enumerable: true, configurable: true };
const arrayFromList = (list: any, length: number): any[] => {
  const out: any[] = [];
  for (let i = 0; i < length; i++) {
    LIST.value = list[i];
    Object.defineProperty(out, i, LIST);
  }
  LIST.value = undefined;
  return out;
};

const staticAll = function all(this: any, iterable: any): any {
  const C = this;
  const capability = newPromiseCapability(C);
  let record: any = null;
  try {
    const promiseResolveFn = C.resolve;
    if (typeof promiseResolveFn !== 'function') throw new TypeError('Promise resolve is not callable');
    record = getIteratorRecord(iterable);
    const values: any = Object.create(null);
    let count = 0;
    let remaining = 1;
    for (;;) {
      const value = stepValue(record);
      if (value === DONE) {
        if (--remaining === 0) capability.resolve(arrayFromList(values, count));
        return capability.promise;
      }
      const index = count++;
      values[index] = undefined;
      const nextPromise = promiseResolveFn.call(C, value);
      let alreadyCalled = false;
      const onFulfilled = builtinClosure(function (x: any) {
        if (alreadyCalled) return undefined;
        alreadyCalled = true;
        values[index] = x;
        if (--remaining === 0) return capability.resolve(arrayFromList(values, count));
        return undefined;
      });
      remaining++;
      nextPromise.then(onFulfilled, capability.reject);
    }
  } catch (error) {
    if (record !== null && !record.done && record.iterator !== null) closeIterator(record.iterator);
    capability.reject(error);
    return capability.promise;
  }
};

const staticRace = function race(this: any, iterable: any): any {
  const C = this;
  const capability = newPromiseCapability(C);
  let record: any = null;
  try {
    const promiseResolveFn = C.resolve;
    if (typeof promiseResolveFn !== 'function') throw new TypeError('Promise resolve is not callable');
    record = getIteratorRecord(iterable);
    for (;;) {
      const value = stepValue(record);
      if (value === DONE) return capability.promise;
      promiseResolveFn.call(C, value).then(capability.resolve, capability.reject);
    }
  } catch (error) {
    if (record !== null && !record.done && record.iterator !== null) closeIterator(record.iterator);
    capability.reject(error);
    return capability.promise;
  }
};
